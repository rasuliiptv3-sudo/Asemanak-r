function readTitle(data) {

function parsePlaylistData(data) {
  const episodes = [];
  let nextToken = null;

  const extractLockup = (lv) => {
    let videoId = null;
    let title = '';
    try {
      const sources = lv?.contentImage?.thumbnailViewModel?.image?.sources || [];
      for (const s of sources) {
        if (s?.url) {
          const m = s.url.match(/\/(?:vi|vi_webp)\/([a-zA-Z0-9_-]{6,})\//);
          if (m) { videoId = m[1]; break; }
        }
      }
    } catch {}
    if (!videoId) {
      try {
        const tap = lv?.navigationViewModel?.onTap || {};
        const we = tap?.innertubeCommand?.watchEndpoint || tap?.watchEndpoint;
        if (we?.videoId) videoId = we.videoId;
      } catch {}
    }
    if (!videoId) {
      try {
        const json = JSON.stringify(lv);
        const m = json.match(/"videoId":"([a-zA-Z0-9_-]{6,})"/);
        if (m) videoId = m[1];
      } catch {}
    }
    try { title = lv?.metadata?.lockupMetadataViewModel?.title?.content || ''; } catch {}
    if (videoId) { episodes.push({ videoId, title }); }
  };

  const extractVideo = (videoId, title) => {
    if (!videoId) return;
    episodes.push({ videoId, title: title || '' });
  };

  const processItem = (item) => {
    if (!item || typeof item !== 'object') return;
    if (item.lockupViewModel) { extractLockup(item.lockupViewModel); }
    else if (item.playlistVideoRenderer) {
      const v = item.playlistVideoRenderer;
      extractVideo(v.videoId, (v.title?.runs || []).map(r => r.text).join(''));
    } else if (item.playlistPanelVideoRenderer) {
      const v = item.playlistPanelVideoRenderer;
      const rawTitle = v.title?.simpleText || (v.title?.runs || []).map(r => r.text).join('');
      extractVideo(v.videoId, rawTitle);
    }
    const tok =
      item?.continuationItemRenderer?.continuationEndpoint?.continuationCommand?.token ||
      item?.continuationItemRenderer?.button?.buttonRenderer?.command?.continuationCommand?.token;
    if (tok) nextToken = tok;
  };

  const visit = (obj) => {
    if (!obj || typeof obj !== 'object') return;
    if (Array.isArray(obj)) { obj.forEach(visit); return; }
    if (Array.isArray(obj.continuationItems)) obj.continuationItems.forEach(processItem);
    if (obj.appendContinuationItemsAction?.continuationItems) obj.appendContinuationItemsAction.continuationItems.forEach(processItem);
    if (obj.itemSectionRenderer) {

    }
    if (obj.playlistVideoListRenderer) {
      (obj.playlistVideoListRenderer.contents || []).forEach(processItem);
    }
    for (const k of Object.keys(obj)) {
      if (k === 'continuationItems') continue;
      visit(obj[k]);
    }
  };
  visit(data);
  return { title: readTitle(data), episodes, nextToken };
}

    const initData = await initRes.json();

    const { title, episodes: initEps, nextToken } = parsePlaylistData(initData);
    const all = [...initEps];
    let token = nextToken;
    let pages = 0;

    while (token && pages < 200) {
      pages++;
      let cjson = null;
      try {
        const cbody = JSON.stringify({ context: CONTEXT, continuation: token });
        const cres = await fetch(BROWSE_URL, { method: 'POST', headers: HEADERS, body: cbody });
        cjson = await cres.json();
      } catch (e) { break; }
      const { episodes: cEps, nextToken: ntok } = parsePlaylistData(cjson);
      all.push(...cEps);
      token = ntok;
      if (cEps.length === 0 && !ntok) break;
    }

    const seen = new Set();
    const dedup = [];
    for (const ep of all) {
      if (!seen.has(ep.videoId)) {
        seen.add(ep.videoId);
        dedup.push(ep);
      }
    }

export function checkIsTV() {
  try { if (document.documentElement.classList.contains('is-tv')) return true; } catch {}
  try {
    const flag = localStorage.getItem("ayeneh_force_tv");
    if (flag === "1") return true;
    if (flag === "0") return false;
  } catch {}
  try {
    const params = new URLSearchParams(window.location.search);
    if (params.get("tv") === "1") return true;
    if (params.get("tv") === "0") return false;
  } catch {}
  const ua = navigator.userAgent || "";
  if (/tv|smarttv|google ?tv|netcast|baidutv|crkey|smart-tv|googletv|webos|tizen|bravia|firetv|fire-tv|aftt|aftm|aftb|aftd|shield|leanback|onn/i.test(ua)) return true;
  try {
    const hasTouch = ('ontouchstart' in window) || (navigator.maxTouchPoints > 0);
    const isAndroid = /android/i.test(ua);
    if (isAndroid && !hasTouch) return true;
  } catch {}
  try {
    const hasTouch = ('ontouchstart' in window) || (navigator.maxTouchPoints > 0);
    const isCoarse = window.matchMedia && window.matchMedia('(pointer: coarse)').matches;
    const sw = (window.screen && window.screen.width) || 0;
    const sh = (window.screen && window.screen.height) || 0;
    if (!hasTouch && isCoarse && (sw >= 1280 || sh >= 1280 || window.innerWidth >= 1000 || window.innerHeight >= 1000)) return true;
  } catch {}
  try {
    const hasTouch = ('ontouchstart' in window) || (navigator.maxTouchPoints > 0);
    if (window.innerWidth < 768 && hasTouch) return false;
  } catch {}
  return false;
}

import { useState, useEffect, useRef } from "react";

export function usePullToRefresh({ onRefresh, disabled, threshold = 70, containerRef }) {
  const ownRef = useRef(null);
  const ref = containerRef || ownRef;
  const [pull, setPull] = useState(0);
  const [refreshing, setRefreshing] = useState(false);
  const st = useRef({ startY: null, pulling: false, refreshing: false, lastDy: 0 });

  useEffect(() => {
    const el = ref.current;
    if (!el) return;
    const isMobile = () => window.innerWidth < 768;

    const onStart = (e) => {
      st.current.startY = e.touches[0].clientY;
      st.current.lastDy = 0;
      st.current.pulling = false;
    };
    const onMove = (e) => {
      if (st.current.startY === null || st.current.refreshing) return;
      const dy = e.touches[0].clientY - st.current.startY;
      st.current.lastDy = dy;
      const atTop = el.scrollTop <= 0;
      if (disabled || !isMobile() || !atTop) {
        if (st.current.pulling) { st.current.pulling = false; setPull(0); }
        return;
      }
      if (dy <= 0) { st.current.pulling = false; setPull(0); return; }
      st.current.pulling = true;
      setPull(Math.min(threshold * 1.6, dy * 0.5));
    };
    const onEnd = async () => {
      if (!st.current.pulling) { st.current.startY = null; return; }
      st.current.pulling = false;
      st.current.startY = null;
      if (st.current.lastDy >= threshold && !st.current.refreshing) {
        st.current.refreshing = true;
        setRefreshing(true);
        setPull(threshold);
        try { await onRefresh?.(); } finally {
          st.current.refreshing = false;
          setRefreshing(false);
          setPull(0);
        }
      } else { setPull(0); }
    };

    el.addEventListener("touchstart", onStart, { passive: true });
    el.addEventListener("touchmove", onMove, { passive: true });
    el.addEventListener("touchend", onEnd, { passive: true });
    return () => {
      el.removeEventListener("touchstart", onStart);
      el.removeEventListener("touchmove", onMove);
      el.removeEventListener("touchend", onEnd);
    };
  }, [ref, disabled, threshold, onRefresh]);

  return { ref, pull, refreshing };
}

const DB_NAME = "ayeneh_db";
const DB_VERSION = 2;
const STORES = ["channels", "series", "meta", "videos"];

let dbPromise = null;

function openDB() {
  if (dbPromise) return dbPromise;
  dbPromise = new Promise((resolve, reject) => {
    try {
      const req = indexedDB.open(DB_NAME, DB_VERSION);
      req.onupgradeneeded = () => {
        const db = req.result;
        STORES.forEach((s) => { if (!db.objectStoreNames.contains(s)) db.createObjectStore(s); });
      };
      req.onsuccess = () => resolve(req.result);
      req.onerror = () => reject(req.error);
    } catch (e) { reject(e); }
  });
  return dbPromise;
}

export async function idbGet(store, key) {
  try {
    const db = await openDB();
    return await new Promise((resolve) => {
      const tx = db.transaction(store, "readonly");
      const req = tx.objectStore(store).get(key);
      req.onsuccess = () => resolve(req.result || null);
      req.onerror = () => resolve(null);
    });
  } catch { return null; }
}

export async function idbSet(store, key, value) {
  try {
    const db = await openDB();
    return await new Promise((resolve) => {
      const tx = db.transaction(store, "readwrite");
      tx.objectStore(store).put(value, key);
      tx.oncomplete = () => resolve(true);
      tx.onerror = () => resolve(false);
    });
  } catch { return false; }
}

export async function idbDel(store, key) {
  try {
    const db = await openDB();
    return await new Promise((resolve) => {
      const tx = db.transaction(store, "readwrite");
      tx.objectStore(store).delete(key);
      tx.oncomplete = () => resolve(true);
      tx.onerror = () => resolve(false);
    });
  } catch { return false; }
}

import { idbGet, idbSet, idbDel } from "@/lib/idb";

const STORE = "videos";
const INDEX_KEY = "ayeneh_offline_index";
const urlCache = new Map();

function readIndex() {
  try { return JSON.parse(localStorage.getItem(INDEX_KEY) || "[]"); } catch { return []; }
}
function writeIndex(list) {
  localStorage.setItem(INDEX_KEY, JSON.stringify(list));
  window.dispatchEvent(new Event("ayeneh-offline-changed"));
}

export function getOfflineIndex() { return readIndex(); }

export async function getOfflineVideoUrl(fileUrl) {
  if (urlCache.has(fileUrl)) return urlCache.get(fileUrl);
  const blob = await idbGet(STORE, fileUrl);
  if (!blob) return null;
  const objUrl = URL.createObjectURL(blob);
  urlCache.set(fileUrl, objUrl);
  return objUrl;
}

export function getOfflineVideoUrlSync(fileUrl) {
  return urlCache.has(fileUrl) ? urlCache.get(fileUrl) : null;
}

export async function isVideoOffline(fileUrl) {
  if (urlCache.has(fileUrl)) return true;
  const blob = await idbGet(STORE, fileUrl);
  return !!blob;
}

export async function cacheVideoOffline(fileUrl, meta = {}, directUrl) {
  const url = directUrl || fileUrl;
  const res = await fetch(url);
  if (!res.ok) throw new Error("fetch_failed");
  const blob = await res.blob();
  await idbSet(STORE, fileUrl, blob);
  const list = readIndex().filter(e => e.fileUrl !== fileUrl);
  list.push({ fileUrl, label: meta.label || fileUrl, size: blob.size, cachedAt: Date.now() });
  writeIndex(list);
  return true;
}

export async function removeOfflineVideo(fileUrl) {
  if (urlCache.has(fileUrl)) { URL.revokeObjectURL(urlCache.get(fileUrl)); urlCache.delete(fileUrl); }
  await idbDel(STORE, fileUrl);
  writeIndex(readIndex().filter(e => e.fileUrl !== fileUrl));
}

export async function clearAllOffline() {
  const list = readIndex();
  for (const e of list) {
    if (urlCache.has(e.fileUrl)) { URL.revokeObjectURL(urlCache.get(e.fileUrl)); urlCache.delete(e.fileUrl); }
    await idbDel(STORE, e.fileUrl);
  }
  writeIndex([]);
}

}

const MEDIA_FAV_KEY = "ayeneh_media_favorites";

function isMediaFav(id) {
  try { return JSON.parse(localStorage.getItem(MEDIA_FAV_KEY) || "[]").includes(id); } catch { return false; }
}
function toggleMediaFav(id) {
  try {
    const list = JSON.parse(localStorage.getItem(MEDIA_FAV_KEY) || "[]");
    const next = list.includes(id) ? list.filter(x => x !== id) : [...list, id];
    localStorage.setItem(MEDIA_FAV_KEY, JSON.stringify(next));
    window.dispatchEvent(new Event("ayeneh-media-favorites-changed"));
    return !list.includes(id);
  } catch { return false; }
}

function parsePosterPos(pos) {
  if (!pos) return { x: 50, y: 20 };
  const parts = pos.trim().split(/\s+/);
  const parseVal = (v) => {
    if (v === "center") return 50;
    if (v === "top" || v === "left") return 0;
    if (v === "bottom" || v === "right") return 100;
    return parseInt(v) || 50;
  };
  return { x: parts[0] ? parseVal(parts[0]) : 50, y: parts[1] ? parseVal(parts[1]) : 20 };
}

            <button onClick={() => { const v = document.querySelector('video'); if (v) { if (v.paused) v.play(); else v.pause(); } }} className="w-12 h-12 rounded-full flex items-center justify-center transition-all hover:scale-110" style={{ background: "rgba(255,255,255,0.15)", border: "1px solid rgba(255,255,255,0.2)", backdropFilter: "blur(8px)" }}>{isPlaying ? <Pause className="w-5 h-5 text-white" /> : <Play className="w-5 h-5 text-white" />}</button>
            <button onClick={() => { const v = document.querySelector('video'); if (v) { v.muted = !v.muted; setMuted(v.muted); } }} className="w-10 h-10 rounded-full flex items-center justify-center transition-all hover:scale-110" style={{ background: "rgba(255,255,255,0.1)", border: "1px solid rgba(255,255,255,0.15)", backdropFilter: "blur(8px)" }}>{muted ? <VolumeX className="w-4 h-4 text-white/70" /> : <Volume2 className="w-4 h-4 text-white" />}</button>
            <button onClick={onSkip} className="flex items-center gap-2 px-5 py-2.5 rounded-lg transition-all hover:scale-105 active:scale-95" style={{ background: "transparent", border: "1.5px solid rgba(255,255,255,0.8)", boxShadow: "0 0 15px rgba(255,255,255,0.1)" }}><span className="text-white font-bold text-xs uppercase tracking-wider whitespace-nowrap">Skip Trailer</span><Play className="w-3.5 h-3.5 fill-white text-white" /></button>
          </div>
        </div>
      </div>
    );
  }

  return (
    <div className="fixed inset-0 z-[90] bg-black flex items-center justify-center">
      <div className="absolute inset-0 overflow-hidden">
        <div style={{ position: "absolute", top: "50%", left: "50%", transform: "translate(-50%, -50%)", width: "100vw", height: "56.25vw", minHeight: "100%", minWidth: "177.78vh", pointerEvents: "none" }}>
          <div ref={containerRef} className="w-full h-full" />
        </div>
      </div>
      <div className="absolute inset-0" style={{ background: "rgba(0,0,0,0.35)" }} />
      <div className="absolute inset-0 z-[5]" />
      <button onClick={onClose} className="absolute top-4 right-4 z-20 w-10 h-10 rounded-full flex items-center justify-center transition-all hover:scale-110" style={{ background: "rgba(0,0,0,0.6)", border: "1px solid rgba(255,255,255,0.2)", backdropFilter: "blur(8px)" }}><X className="w-5 h-5 text-white" /></button>
      <div className="absolute top-4 left-4 z-20 flex items-center gap-2">
        <div className="w-9 h-9 rounded-full overflow-hidden flex-shrink-0" style={{ boxShadow: "0 0 10px rgba(123,44,191,0.4)", border: "1.5px solid rgba(123,44,191,0.35)" }}><img src={LOGO_URL} alt="" className="w-full h-full object-cover" draggable={false} /></div>
        <div className="leading-tight"><p className="text-white font-bold text-sm">آسمانک</p><p className="text-purple-300/70 text-[9px]">{series?.title}</p></div>
      </div>
      {!ready && (<div className="absolute inset-0 flex items-center justify-center z-[15]"><div className="w-8 h-8 border-3 border-purple-500/30 border-t-purple-500 rounded-full animate-spin" /></div>)}
      <div className="absolute bottom-0 left-0 right-0 z-20 px-4 pb-6 pt-20" style={{ background: "linear-gradient(180deg, transparent 0%, rgba(0,0,0,0.9) 50%)" }}>
        <div className="mb-4"><span className="text-xs font-bold uppercase tracking-widest text-purple-400 mb-1 block">تریلر • Trailer</span><h2 className="text-white font-black text-xl md:text-2xl">{series?.title}</h2></div>
        <div className="flex items-center justify-center gap-4">
          <button onClick={togglePlay} className="w-12 h-12 rounded-full flex items-center justify-center transition-all hover:scale-110" style={{ background: "rgba(255,255,255,0.15)", border: "1px solid rgba(255,255,255,0.2)", backdropFilter: "blur(8px)" }}>{isPlaying ? <Pause className="w-5 h-5 text-white" /> : <Play className="w-5 h-5 text-white" />}</button>
          <button onClick={toggleMute} className="w-10 h-10 rounded-full flex items-center justify-center transition-all hover:scale-110" style={{ background: "rgba(255,255,255,0.1)", border: "1px solid rgba(255,255,255,0.15)", backdropFilter: "blur(8px)" }}>{muted ? <VolumeX className="w-4 h-4 text-white/70" /> : <Volume2 className="w-4 h-4 text-white" />}</button>
          <button onClick={onSkip} className="flex items-center gap-2 px-5 py-2.5 rounded-lg transition-all hover:scale-105 active:scale-95" style={{ background: "transparent", border: "1.5px solid rgba(255,255,255,0.8)", boxShadow: "0 0 15px rgba(255,255,255,0.1)" }}><span className="text-white font-bold text-xs uppercase tracking-wider whitespace-nowrap">Skip Trailer</span><Play className="w-3.5 h-3.5 fill-white text-white" /></button>
        </div>
      </div>
    </div>
  );
}
